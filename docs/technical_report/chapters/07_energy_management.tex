\chapter{Energy Management Model}
\label{ch:energy}

\section{Energy Balance}

At every time step $t$ with duration $\Delta t = 0.25$~h (15 minutes):

\begin{equation}
    P_{\text{PV}}(t) + P_{\text{grid}}(t) + P_{\text{dis}}(t) = P_{\text{load}}(t) + P_{\text{ch}}(t) + P_{\text{curtail}}(t)
    \label{eq:balance}
\end{equation}

where all terms are in watts and represent instantaneous power. The constraint $P_{\text{grid}}(t) \geq 0$ enforces no grid export.

\section{Battery Model}

The battery state of charge evolves according to Equation~\ref{eq:soc}. The operating constraints are:

\begin{align}
    \text{SOC}_{\min} &\leq \text{SOC}(t) \leq \text{SOC}_{\max} & \forall t \label{eq:soc_bounds} \\
    0 &\leq P_{\text{ch}}(t) \leq P_{\text{ch,max}} & \forall t \label{eq:ch_limit} \\
    0 &\leq P_{\text{dis}}(t) \leq P_{\text{dis,max}} & \forall t \label{eq:dis_limit}
\end{align}

Simultaneous charging and discharging is prevented using a binary variable $z(t) \in \{0, 1\}$:

\begin{align}
    P_{\text{ch}}(t) &\leq P_{\text{ch,max}} \cdot z(t) \\
    P_{\text{dis}}(t) &\leq P_{\text{dis,max}} \cdot (1 - z(t))
\end{align}
